idf_build_get_property(g1_only BAJI_G1_ONLY)
idf_build_get_property(character BAJI_CHARACTER)
idf_build_get_property(wireless BAJI_WIRELESS)
idf_build_get_property(frame_probe BAJI_FRAME_PROBE)
set(app_requires baji_board esp_timer)
if(wireless AND NOT g1_only)
    list(APPEND app_requires baji_connectivity)
endif()
if(NOT g1_only)
    if(character STREQUAL "GPT")
        list(APPEND app_requires baji_pet_app)
    else()
        list(APPEND app_requires baji_deepseek)
    endif()
endif()
idf_component_register(SRCS "app_main.c" PRIV_REQUIRES ${app_requires} INCLUDE_DIRS "")
if(frame_probe AND NOT g1_only)
    target_sources(${COMPONENT_LIB} PRIVATE
        "../tests/device/product_frames.c" "../tests/device/deepseek_review/main/body_probe.c")
    target_link_libraries(${COMPONENT_LIB} INTERFACE
        "-Wl,--wrap=baji_board_paint_round_regions" "-Wl,-u,__wrap_baji_board_paint_round_regions")
endif()
if(wireless AND NOT g1_only)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE BAJI_WIRELESS=1)
endif()
if(g1_only)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE BAJI_G1_ONLY=1)
elseif(character STREQUAL "GPT")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE BAJI_GPT_PET=1)
endif()
if(g1_only OR CONFIG_BAJI_BOARD_CHECK)
    target_sources(${COMPONENT_LIB} PRIVATE "board_check.c")
endif()
