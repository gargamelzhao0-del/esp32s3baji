# The following lines of boilerplate have to be in your project's
# CMakeLists in this exact order for cmake to work correctly
cmake_minimum_required(VERSION 3.22)
set(PROJECT_VER "0.1.0")
option(BAJI_WIRELESS "Enable the Wi-Fi and BLE agent bridge" ON)
option(BAJI_FRAME_PROBE "Measure visible character changes at the display painter" OFF)
option(BAJI_INTERACTION_PROBE "Replay GPT touch gestures on the board for validation" OFF)
if(BAJI_G1_ONLY)
    set(BAJI_WIRELESS OFF)
endif()
if(BAJI_WIRELESS)
    if(NOT SDKCONFIG_DEFAULTS)
        set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_SOURCE_DIR}/sdkconfig.defaults")
    endif()
    list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_SOURCE_DIR}/config/sdkconfig.wireless.defaults")
endif()

# Keep the old preparation sdkconfig and other build profiles isolated.
if(NOT DEFINED SDKCONFIG)
    set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
# "Trim" the build. Include the minimal set of components, main, and anything it depends on.
idf_build_set_property(MINIMAL_BUILD ON)
idf_build_set_property(BAJI_G1_ONLY "${BAJI_G1_ONLY}")
idf_build_set_property(BAJI_WIRELESS "${BAJI_WIRELESS}")
idf_build_set_property(BAJI_FRAME_PROBE "${BAJI_FRAME_PROBE}")
idf_build_set_property(BAJI_INTERACTION_PROBE "${BAJI_INTERACTION_PROBE}")
set(BAJI_CHARACTER "DeepSeek" CACHE STRING "Default badge character (DeepSeek or GPT)")
set_property(CACHE BAJI_CHARACTER PROPERTY STRINGS DeepSeek GPT)
if(NOT BAJI_CHARACTER MATCHES "^(DeepSeek|GPT)$")
    message(FATAL_ERROR "BAJI_CHARACTER must be DeepSeek or GPT")
endif()
idf_build_set_property(BAJI_CHARACTER "${BAJI_CHARACTER}")
project(esp32s3baji)

# Keep the large, independently indexed HD frames out of the executable's
# read-only segment. The normal IDF flash target installs both images.
if(NOT BAJI_G1_ONLY AND NOT CONFIG_BAJI_BOARD_CHECK)
    if(BAJI_CHARACTER STREQUAL "GPT")
        set(motion_file "${CMAKE_CURRENT_SOURCE_DIR}/components/baji_pet_clips/assets/clips.bpc")
        set(pack_name "clips.bpc")
    else()
        set(motion_file "${CMAKE_CURRENT_SOURCE_DIR}/components/baji_deepseek_hd/assets/motion.hda")
        set(pack_name "hdmotion.hda")
    endif()
    partition_table_get_partition_info(motion_capacity "--partition-name hdmotion" "size")
    if(NOT motion_capacity)
        message(FATAL_ERROR "Missing hdmotion data partition")
    endif()
    file(SIZE "${motion_file}" motion_size)
    math(EXPR motion_capacity "${motion_capacity}")
    if(motion_size GREATER motion_capacity)
        message(FATAL_ERROR "The character animation pack exceeds its data partition")
    endif()
    configure_file("${motion_file}" "${CMAKE_BINARY_DIR}/${pack_name}" COPYONLY)
    esptool_py_flash_to_partition(flash "hdmotion" "${CMAKE_BINARY_DIR}/${pack_name}")
endif()
